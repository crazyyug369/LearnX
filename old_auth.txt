/**
 * Server API: Authenticate or register a student, retrieving persistent progress from MySQL
 */
export async function apiLoginUser(params: {
  email?: string;
  name?: string;
  role?: UserRole;
  grade?: string;
  interest?: StudentInterest;
  studentId?: string;
  institution?: string;
  institutionId?: string;
  institutionCode?: string;
  department?: string;
  teacherId?: string;
  password?: string;
}): Promise<{
  success: boolean;
  user: UserProfile;
  conceptsMap: Record<string, ConceptNode[]>;
  overallMastery: number;
  isNew: boolean;
}> {
  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      credentials: 'include',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(params),
    });

    if (res.ok) {
      const data = await res.json();
      if (data.success && data.user) {
        // Cache user concepts locally for instant offline/re-render resilience
        if (data.conceptsMap) {
          localStorage.setItem(
            `${STUDENT_CONCEPTS_PREFIX}${data.user.id}`,
            JSON.stringify(data.conceptsMap)
          );
        }
        return data;
      }
    }
  } catch (err) {
    console.warn('MySQL login endpoint unreachable, falling back to local storage:', err);
  }

  // Fallback if backend server is temporarily starting up
  const cleanName = params.name?.trim() || params.email?.split('@')[0] || 'Learner';
  const slug = cleanName.toLowerCase().replace(/[^a-z0-9]/g, '_');
  const userId = `stu_${slug}`;
  const localConcepts = getStudentConcepts(userId, userId, cleanName);

  const localUser: UserProfile = {
    id: userId,
    name: cleanName,
    email: params.email || `${cleanName.toLowerCase()}@student.learnx.org`,
    role: params.role || 'student',
    grade: params.grade || 'Grade 10',
    interest: params.interest || 'Cricket & Sports',
    studentId: params.studentId || `STU-2026-${Math.floor(1000 + Math.random() * 9000)}`,
    streakCount: 1,
    avatar: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=100&auto=format&fit=crop&q=80',
  };

  return {
    success: true,
    user: localUser,
    conceptsMap: localConcepts,
    overallMastery: calculateOverallMastery(localConcepts),
    isNew: false,
  };
}
